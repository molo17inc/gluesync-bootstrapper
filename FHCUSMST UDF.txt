	/*
        ''' <summary>
        ''' Converts nullable DB2 MM, DD, YY or YYYY decimal fields to nullable MSSQL DateTime
        ''' field.
        ''' </summary>
        ''' <param name="formatYear">Format pattern indicating the size of the year.</param>
        ''' <param name="month">Decimal value for month part of the new date.</param>
        ''' <param name="day">Decimal value for day part of the new date.</param>
        ''' <param name="year">Decimal value for year part of the new date.  Either YY or YYYY format.</param>
        ''' <exception cref="ArgumentOutOfRangeException">
        ''' <paramref name="formatYear"/> is not "YY" or "YYYY".
        ''' </exception>
        ''' <returns>Nullable DateTime with validation rules for SQL Server DateTime type</returns>
        ''' <remarks>Use this function to make an SQL Server datetime from individual
        ''' DB2 decimal date parts.  The size of the year fields in DB2 will vary, so
        '''  the formatYear parameter is required to distinguish long or short year.</remarks>
	*/
	public static Object makeDateTime(String formatYear, BigDecimal month, BigDecimal day, BigDecimal year) throws Exception{
	  int fullYear;
	  int intMonth = month.intValue();
	  int intDay = day.intValue();
	  int intYear = year.intValue();

	  if(formatYear.toUpperCase() == "YY"){
	    if(intYear <= 39){
	      fullYear = 2000 + intYear;
	    }else{
	      fullYear = 1900 + intYear;
	    }
	  } else if(formatYear.toUpperCase() == "YYYY"){
	    fullYear = year.intValue();
          } else{
	    throw new Exception("Valid year formats are YY or YYYY");
          }
	  /*
            ' 30 days has September, April, June and November all the rest     
            ' have 31 except for February which is fine it has 28 days
            ' although leap year makes it 29.  An imperfect, but "close enough“     
            ' way to calculate leap years is when the following is true:
            '      0 == year (mod 4)
	  */
	  /* Validate the months */
	  if((intMonth < 1) || (intMonth > 12)){
	    return null;
	  }
	  /* No months have a day less than 1 or greater than 31 */
	  if((intDay < 1) || (intDay > 31)){
	    return null;
	  }
	  /* The low datetime value in MSSQL 1/1/1753 */
	  if(fullYear < 1753){
	    return null;
	  }
	  /* Validate the days per month */
	  if(((intMonth == 4) || (intMonth == 6) || (intMonth == 9) || (intMonth == 11)) && (intDay > 30)){
	    return null;
	  } else if(intMonth == 2){ /* February is a special case */
	    /* Leap year */
	    if((fullYear % 4) == 0){
	      if(intDay > 29){
	        return null;
	      }
	    } else {
	      /* Normal year */
	      if(intDay > 28){
	        return null;
	      }
	    }
	  }

	  /* If we're here, then we have a valid date.  Now convert the passed parameters to an actual date. */
	  return LocalDate.of(fullYear, intMonth, intDay);
	}

	  int seconds = tempTime % 100;

	  if((hours > 23) || (minutes > 59) || (seconds > 59)){
	    return fullDate;
	  }

	  return LocalDateTime.of(fullDate, LocalTime.of(hours, minutes, seconds));
	}
